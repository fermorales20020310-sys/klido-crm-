async function forgot(){ let email=prompt('Escribe tu CORREO registrado (ej: tu@gmail.com):'); if(!email) return; try{ let r=await fetch('/api/forgot-password',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({email})}); let j=await r.json(); if(r.ok) alert('Te enviamos correo a '+email+'\nRevisa bandeja y SPAM.\nAgencias: '+(j.agencies||[]).join(', ')+'\nLink dura 15 min'); else alert(j.error||'Error'); }catch(e){ alert('Error de conexión'); } }
